function renderQR(){
  const n=11,cells=[];
  for(let y=0;y<n;y++){
    for(let x=0;x<n;x++){
      const inCorner=(x<3&&y<3)||(x>n-4&&y<3)||(x<3&&y>n-4);
      const cornerDot=inCorner&&((x+y)%2===0||(x===1&&y===1)||(x===n-2&&y===1)||(x===1&&y===n-2));
      const on=inCorner?cornerDot:((x*7+y*5+pairSeed)%3===0);
      cells.push('<i class="'+(on?'':'off')+'"></i>');
    }
  }
  document.getElementById('qrBox').innerHTML=cells.join('');
}

function startTransfer(){
  const items=pairs.filter(p=>p.ok&&picked.has(p.id));
  if(!items.length){toast('先在映射表里勾选要互传的内容');return;}
  if(xferBusy)return;
  xferBusy=true;
  document.getElementById('xferSub').textContent='传输中 · '+srcName()+' → '+(xferDir==='down'?'电脑':'手机')+' · '+items.length+' 项';
  document.getElementById('xferList').innerHTML=items.map((p,i)=>`
    <div class="xfer" data-i="${i}">
      <div class="xfer-dir ${xferDir==='down'?'':'up'}"><svg viewBox="0 0 24 24">${xferDir==='down'
        ?'<path d="M12 4v14m0 0l-5-5m5 5l5-5"/>':'<path d="M12 20V6m0 0l-5 5m5-5l5 5"/>'}</svg></div>
      <div class="xfer-main"><div class="xfer-t">${esc(p.id==='instances'?instances[selected].name+' 实例目录':p.id)}</div>
        <div class="xfer-s">${esc(p.phone)} · ${esc(p.size)}</div>
        <div class="progress" style="width:100%;margin-top:7px"><i data-bar="${i}" style="width:0%"></i></div></div>
      <div class="xfer-right"><div data-pct="${i}">0%</div><div data-sp="${i}">—</div></div>
    </div>`).join('');
  const prog=items.map(()=>0), done=items.map(()=>false);
  clearInterval(xferTimer);
  xferTimer=setInterval(()=>{
    let all=true;
    items.forEach((p,i)=>{
      if(done[i])return;
      all=false;
      prog[i]=Math.min(100,prog[i]+(p.size.startsWith('4.8')?3:p.size.startsWith('1.2')?5:9)+Math.random()*6);
      const bar=document.querySelector(`[data-bar="${i}"]`),pct=document.querySelector(`[data-pct="${i}"]`),sp=document.querySelector(`[data-sp="${i}"]`);
      if(!bar)return;
      bar.style.width=prog[i]+'%';pct.textContent=Math.floor(prog[i])+'%';
      sp.textContent=(6+Math.random()*7).toFixed(1)+' MB/s';
      if(prog[i]>=100){done[i]=true;bar.parentElement.style.opacity=.5;pct.textContent='完成';pct.style.color='var(--accent)';
        logLine('<span class="lt">'+stamp()+'</span> <span class="lok">[OK]</span> '+esc(p.id)+' 传输完成（'+esc(p.size)+'）');}
    });
    const cur=items[Math.min(items.length-1,done.findIndex(d=>!d)<0?items.length-1:done.findIndex(d=>!d))];
    document.getElementById('devSpeed').textContent=(6+Math.random()*7).toFixed(1)+' MB/s';
    if(all){
      clearInterval(xferTimer);xferBusy=false;
      document.getElementById('xferSub').textContent='已完成 · '+items.length+' 项 · 校验通过';
      document.getElementById('devSpeed').textContent='—';
      toast('互传完成 · '+items.length+' 项已'+(xferDir==='down'?'落到电脑':'推到手机')+'公共目录');
      logLine('<span class="lt">'+stamp()+'</span> <span class="lok">[DONE]</span> 队列完成，manifest 校验一致');
      document.getElementById('sbDevTxt').textContent='手机已连接 · 刚互传 '+items.length+' 项';
    }
  },500);
  drawer.classList.add('open');
  logLine('<span class="lt">'+stamp()+'</span> 开始互传：'+srcName()+' → '+(xferDir==='down'?'电脑':'手机')+'，共 '+items.length+' 项，策略「'+policy+'」');
}

/* ================= 三栏宽度可拖拽 ================= */
let navW=200,sideW=320;
function applyW(){
  winEl.style.setProperty('--nav-w',navW+'px');
  winEl.style.setProperty('--side-w',sideW+'px');
}
document.querySelectorAll('.grip').forEach(g=>{
  g.addEventListener('pointerdown',e=>{
    if(winEl.classList.contains('narrow')&&g.dataset.grip==='nav'){toast('侧栏已收起，先点击"收起侧栏"展开');return;}
    const edge=g.dataset.grip,sx=e.clientX,sw=g.dataset.grip==='nav'?navW:sideW;
    g.classList.add('drag');document.body.classList.add('resizing');
    g.setPointerCapture(e.pointerId);
    const move=ev=>{
      const d=ev.clientX-sx;
      if(edge==='nav')navW=Math.min(320,Math.max(150,sw+d));else sideW=Math.min(430,Math.max(260,sw-d));
      applyW();
    };
    const up=()=>{
      g.classList.remove('drag');document.body.classList.remove('resizing');
      g.removeEventListener('pointermove',move);g.removeEventListener('pointerup',up);
      toast((edge==='nav'?'导航栏':'操作栏')+'宽度 '+Math.round(edge==='nav'?navW:sideW)+' px');
    };
    g.addEventListener('pointermove',move);g.addEventListener('pointerup',up);
  });
  g.addEventListener('dblclick',()=>{
    if(g.dataset.grip==='nav')navW=200;else sideW=320;applyW();toast('已恢复默认栏宽');
  });
});

/* 快照版（25w43a）与 1.20.5+ 走 JDK 21，1.17–1.20.4 走 17，其余走 8 */
function javaIdx(v){
  if(/^\d{2}w\d{2}[a-z]$/.test(v))return 0;
  const mm=/^b?1\.(\d+)/.exec(v);const minor=mm?+mm[1]:99;
  if(minor>=20)return /\.(5|6|7|8|9)$/.test(v)?0:1;
  if(minor>=17)return 1;
  if(v==='b1.7.3'||v==='1.5.2')return 2;
  return 2;
}
